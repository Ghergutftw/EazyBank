🚀 ACCESSING YOUR SERVICES
================================================================================

{{- if .Values.gatewayserver.enabled }}

1️⃣  Gateway Server (Main API Entry Point):
{{- if eq .Values.gatewayserver.service.type "LoadBalancer" }}

   ⏳ Waiting for LoadBalancer IP (may take 1-2 minutes)...

   Windows (PowerShell):
   $GATEWAY_IP = kubectl get svc gatewayserver -n {{ .Release.Namespace }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}'
   Write-Host "Gateway URL: http://$GATEWAY_IP:{{ .Values.gatewayserver.service.port }}"

   Linux/Mac:
   export GATEWAY_IP=$(kubectl get svc gatewayserver -n {{ .Release.Namespace }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
   echo "Gateway URL: http://$GATEWAY_IP:{{ .Values.gatewayserver.service.port }}"

   Test APIs:
   curl http://$GATEWAY_IP:{{ .Values.gatewayserver.service.port }}/eazybank/accounts/api/contact-info
   curl http://$GATEWAY_IP:{{ .Values.gatewayserver.service.port }}/eazybank/loans/api/contact-info
   curl http://$GATEWAY_IP:{{ .Values.gatewayserver.service.port }}/eazybank/cards/api/contact-info

{{- else if eq .Values.gatewayserver.service.type "NodePort" }}

   Windows (PowerShell):
   $NODE_PORT = kubectl get svc gatewayserver -n {{ .Release.Namespace }} -o jsonpath='{.spec.ports[0].nodePort}'
   $NODE_IP = kubectl get nodes -o jsonpath='{.items[0].status.addresses[0].address}'
   Write-Host "Gateway URL: http://$NODE_IP:$NODE_PORT"

   Linux/Mac:
   export NODE_PORT=$(kubectl get svc gatewayserver -n {{ .Release.Namespace }} -o jsonpath='{.spec.ports[0].nodePort}')
   export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[0].address}')
   echo "Gateway URL: http://$NODE_IP:$NODE_PORT"

{{- else }}

   Port-forward to access locally:
   kubectl port-forward svc/gatewayserver {{ .Values.gatewayserver.service.port }}:{{ .Values.gatewayserver.service.port }} -n {{ .Release.Namespace }}

   Then access: http://localhost:{{ .Values.gatewayserver.service.port }}

   Test APIs:
   curl http://localhost:{{ .Values.gatewayserver.service.port }}/eazybank/accounts/api/contact-info
{{- end }}
{{- end }}

📝 USEFUL KUBERNETES COMMANDS

Monitor Deployment Status:
  kubectl get pods -n {{ .Release.Namespace }} -w
  kubectl get deployments -n {{ .Release.Namespace }}
  kubectl get services -n {{ .Release.Namespace }}
  kubectl get all -n {{ .Release.Namespace }}

View Logs:
  # Follow logs for a service
  kubectl logs -f deployment/accounts -n {{ .Release.Namespace }}

  # View logs from all containers
  kubectl logs -f deployment/accounts --all-containers -n {{ .Release.Namespace }}

  # View previous logs (if pod crashed)
  kubectl logs deployment/accounts --previous -n {{ .Release.Namespace }}

Debug Pod Issues:
  # Describe a pod (shows events and errors)
  kubectl describe pod <pod-name> -n {{ .Release.Namespace }}

  # Get pod events
  kubectl get events --sort-by='.lastTimestamp' -n {{ .Release.Namespace }}

  # Execute commands in a pod
  kubectl exec -it deployment/accounts -n {{ .Release.Namespace }} -- sh

Check Configuration:
  # View ConfigMap
  kubectl get configmap eazybank-config -n {{ .Release.Namespace }} -o yaml

  # Check environment variables
  kubectl exec deployment/accounts -n {{ .Release.Namespace }} -- env | grep SPRING

Scale Microservices:
  # Scale to 3 replicas
  kubectl scale deployment accounts --replicas=3 -n {{ .Release.Namespace }}

  # Scale multiple services
  kubectl scale deployment accounts loans cards --replicas=2 -n {{ .Release.Namespace }}

Resource Usage:
  # View resource consumption
  kubectl top pods -n {{ .Release.Namespace }}
  kubectl top nodes

Health Checks:
  # Check health endpoints
  kubectl exec deployment/accounts -n {{ .Release.Namespace }} -- wget -qO- http://localhost:8080/actuator/health
  kubectl exec deployment/loans -n {{ .Release.Namespace }} -- wget -qO- http://localhost:8090/actuator/health
  kubectl exec deployment/cards -n {{ .Release.Namespace }} -- wget -qO- http://localhost:9000/actuator/health
